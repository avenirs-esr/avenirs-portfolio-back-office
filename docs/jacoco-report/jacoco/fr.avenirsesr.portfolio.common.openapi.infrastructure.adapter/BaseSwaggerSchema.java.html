package fr.avenirsesr.portfolio.common.openapi.infrastructure.adapter;

import fr.avenirsesr.portfolio.common.authorization.domain.model.enums.EAuthorization;
import fr.avenirsesr.portfolio.common.authorization.domain.model.enums.EAuthorizationEntityType;
import fr.avenirsesr.portfolio.common.data.domain.model.enums.ESortField;
import fr.avenirsesr.portfolio.common.data.domain.model.enums.ESortOrder;
import fr.avenirsesr.portfolio.common.data.domain.model.enums.EUserCategory;
import fr.avenirsesr.portfolio.common.error.domain.model.enums.EErrorCode;
import fr.avenirsesr.portfolio.common.language.domain.model.enums.ELanguage;
import fr.avenirsesr.portfolio.common.security.accesscontrol.domain.model.enums.ERole;
import fr.avenirsesr.portfolio.common.security.accesscontrol.domain.model.enums.EUniverse;
import fr.avenirsesr.portfolio.common.temporal.domain.model.enums.EDurationUnit;
import io.swagger.v3.oas.models.media.Schema;
import io.swagger.v3.oas.models.media.StringSchema;
import java.util.Arrays;

public class BaseSwaggerSchema {

  public static final Schema<String> errorCodeSchema =
      new StringSchema()
          .name("EErrorCode")
          ._enum(Arrays.stream(EErrorCode.values()).map(Enum::name).toList())
          .description("Enum for error codes");

  public static final Schema<String> languageSchema =
      new StringSchema()
          .name("ELanguage")
          ._enum(Arrays.stream(ELanguage.values()).map(Enum::name).toList())
          .description("Enum for languages");

  public static final Schema<String> sortFieldSchema =
      new StringSchema()
          .name("ESortField")
          ._enum(Arrays.stream(ESortField.values()).map(Enum::name).toList())
          .description("Enum for sort fields");

  public static final Schema<String> sortOrderSchema =
      new StringSchema()
          .name("ESortOrder")
          ._enum(Arrays.stream(ESortOrder.values()).map(Enum::name).toList())
          .description("Enum for sort orders");

  public static final Schema<String> durationUnitSchema =
      new StringSchema()
          .name("EDurationUnit")
          ._enum(Arrays.stream(EDurationUnit.values()).map(Enum::name).toList())
          .description("Enum for duration units");

  public static final Schema<String> roleSchema =
      new StringSchema()
          .name("ERole")
          ._enum(Arrays.stream(ERole.values()).map(Enum::name).toList())
          .description("Enum for user role");

  public static final Schema<String> universeSchema =
      new StringSchema()
          .name("EUniverse")
          ._enum(Arrays.stream(EUniverse.values()).map(Enum::name).toList())
          .description("Enum for universe");

  public static final Schema<String> userCategorySchema =
      new StringSchema()
          .name("EUserCategory")
          ._enum(Arrays.stream(EUserCategory.values()).map(Enum::name).toList())
          .description("Enum for user category");

  public static final Schema<String> authorizationSchema =
      new StringSchema()
          .name("EAuthorization")
          ._enum(Arrays.stream(EAuthorization.values()).map(Enum::name).toList())
          .description("Enum for authorization");

  public static final Schema<String> authorizationEntityTypeSchema =
      new StringSchema()
          .name("EAuthorizationEntityType")
          ._enum(Arrays.stream(EAuthorizationEntityType.values()).map(Enum::name).toList())
          .description("Enum for authorization entity type");

  static Schema<String> createEnumSchema(Class<? extends Enum<?>> enumClass, String description) {
    return new StringSchema()
        .name(enumClass.getSimpleName())
        ._enum(Arrays.stream(enumClass.getEnumConstants()).map(Enum::name).toList())
        .description(description);
  }
}
